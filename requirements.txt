# =============================================================================
# privacy-evaluation/MIA: NetSSM and TrafficLLM attacks
#   All four scripts (netssm_{black,white}box_mia.py,
#   trafficllm_{black,white}box_mia.py) need transformers + torch (already
#   listed above) to load their target checkpoint and score
#   perplexity/loss/gradient-norm. transformers>=4.45 is required to match
#   the torch/diffusers/peft/accelerate combination above -- if you only run
#   the NetSSM/TrafficLLM MIA scripts and skip NetDiffusion's white-box
#   attack, you can relax this to transformers>=4.30 (TrafficLLM's own
#   pin) but you no longer need diffusers/peft/accelerate/torchvision at all.
# =============================================================================

# --- NetSSM MIA only: vendored model code, not a pip package ---

cpm_kernels>=1.0,<2       # ChatGLM2's custom CUDA kernels (quantization path)
